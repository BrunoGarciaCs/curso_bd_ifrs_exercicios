CREATE TABLE artist (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL
);

CREATE TABLE show (
  id INTEGER PRIMARY KEY,
  artist_id INTEGER NOT NULL,
  date_time DATETIME NOT NULL,
  stage TEXT NOT NULL,
  FOREIGN KEY (artist_id) REFERENCES artist (id)
);

CREATE TABLE ticket (
  id INTEGER PRIMARY KEY,
  show_id INTEGER NOT NULL,
  price REAL NOT NULL,
  type TEXT NOT NULL,
  FOREIGN KEY (show_id) REFERENCES show (id)
);

CREATE TABLE customer (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT NOT NULL
);

CREATE TABLE purchase (
  id INTEGER PRIMARY KEY,
  customer_id INTEGER NOT NULL,
  ticket_id INTEGER NOT NULL,
  purchase_date DATETIME NOT NULL,
  FOREIGN KEY (customer_id) REFERENCES customer (id),
  FOREIGN KEY (ticket_id) REFERENCES ticket (id)
);

01. Como listar o número total de compras feitas por cada cliente?
SELECT customer_id, COUNT(*) 
FROM purchase 
GROUP BY customer_id;



02. Como encontrar o preço total dos ingressos vendidos para o show com ID 2?
SELECT SUM(price)
FROM ticket 
WHERE show_id = 2;



03. Como encontrar o preço médio dos ingressos do tipo 'Regular'?
SELECT AVG(price) 
FROM ticket 
WHERE type = 'Regular';



04. Qual é o comando para inserir uma nova compra na tabela 'purchase'?
INSERT INTO purchase (id, customer_id, ticket_id, purchase_date) 
VALUES (6, 2, 7, '2023-07-18 12:00:00');



05. Qual é o comando para alterar a data e hora de um show?
UPDATE show SET date_time = '2023-08-20 18:00:00' 
WHERE id = 1;



06. Como excluir todos os shows que ocorrem antes de '2023-07-17'?
DELETE FROM show 
WHERE date_time < '2023-07-17';



07. Como listar todos os shows e o respectivo número de ingressos vendidos?
SELECT show.id, COUNT(ticket.id) 
FROM show 
JOIN ticket ON show.id = ticket.show_id 
GROUP BY show.id;



08. Como selecionar todos os ingressos vendidos para shows do artista com ID 2?
SELECT ticket.* 
FROM ticket 
JOIN show ON ticket.show_id = show.id 
WHERE show.artist_id = 2;



09. Qual é o comando para encontrar o nome do artista e a data do show para todos os shows no 'Main Stage'?
SELECT artist.name, show.date_time 
FROM show 
JOIN artist ON show.artist_id = artist.id 
WHERE show.stage = 'Main Stage';



10. Como selecionar o nome e email de todos os clientes cujo nome começa com 'Ana'?
SELECT name, email 
FROM customer 
WHERE name LIKE 'Ana%';



11. Como excluir todos os ingressos com preço superior a 200.00?
DELETE FROM ticket 
WHERE price > 200.00;



12. Qual comando insere uma nova compra feita pelo cliente com ID 1 para o ingresso com ID 10?
INSERT INTO purchase (customer_id, ticket_id, purchase_date) V
ALUES (1, 10, CURRENT_TIMESTAMP);



13. Qual comando atualiza o preço do ingresso para '350.0' onde o tipo é 'VIP'?
UPDATE ticket SET type = 'VIP' 
WHERE price = 350.0;


14. Qual comando insere um novo artista na tabela 'artist'?
INSERT INTO artist (id, name)

15. Como encontrar o nome do cliente que comprou o ingresso mais caro?
SELECT customer.name 
FROM customer 
JOIN purchase ON customer.id = purchase.customer_id 
JOIN ticket ON purchase.ticket_id = ticket.id 
ORDER BY ticket.price 
DESC LIMIT 1;



16. Qual comando lista o email de todos os clientes que compraram ingressos VIP?
SELECT customer.email 
FROM customer 
JOIN purchase ON customer.id = purchase.customer_id 
JOIN ticket ON purchase.ticket_id = ticket.id 
WHERE ticket.type = 'VIP';



17. Qual é o comando para contar o número total de shows em cada palco?
SELECT COUNT() FROM show GROUP BY stage;



18. Como encontrar o preço médio dos ingressos para cada tipo?
SELECT type, AVG(price) FROM ticket GROUP BY type;



19. Qual comando atualiza o email de um cliente específico para 'novo.email@example.com'?
UPDATE customer SET email = 'novo.email@example.com' WHERE id = 1;



20. Qual comando seleciona o show mais caro (baseado no preço do ingresso)?
SELECT show_id FROM ticket ORDER BY price DESC LIMIT 1;
